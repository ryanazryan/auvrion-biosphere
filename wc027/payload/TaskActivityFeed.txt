"use client";

import { useRouter } from "next/navigation";
import { useEffect, useRef, useState } from "react";

type Activity = {
  id: string;
  kind: "transition" | "finding";
  occurredAt: string;
  title: string;
  description: string;
  transitionId?: string;
};
type Snapshot = {
  task: { status: string; version: number };
  items: Activity[];
};

/** Poll a strictly Supervisor-scoped projection. Messages are saved findings, not chat. */
export function TaskActivityFeed({
  taskId,
  organizationId,
}: {
  taskId: string;
  organizationId: string;
}) {
  const router = useRouter();
  const [snapshot, setSnapshot] = useState<Snapshot | null>(null);
  const [error, setError] = useState("");
  const lastVersion = useRef<number | null>(null);

  useEffect(() => {
    let disposed = false;
    const controller = new AbortController();
    const load = async () => {
      if (document.visibilityState === "hidden" || disposed) return;
      try {
        const response = await fetch(
          `/api/tasks/${encodeURIComponent(taskId)}/activity?organizationId=${encodeURIComponent(organizationId)}`,
          { cache: "no-store", credentials: "same-origin", signal: controller.signal },
        );
        if (!response.ok) throw new Error(`HTTP_${response.status}`);
        const data = (await response.json()) as Snapshot;
        if (!Array.isArray(data.items) || !data.task) {
          throw new Error("INVALID_ACTIVITY_RESPONSE");
        }
        if (disposed) return;
        setSnapshot(data);
        setError("");
        if (
          lastVersion.current !== null &&
          data.task.version !== lastVersion.current
        ) {
          // Refresh server-rendered task status/buttons when workflow changes.
          router.refresh();
        }
        lastVersion.current = data.task.version;
      } catch (cause) {
        if (disposed || controller.signal.aborted) return;
        setError(
          cause instanceof Error && cause.message === "HTTP_403"
            ? "Akses progres tugas ditolak. Periksa sesi Supervisor."
            : "Progres belum dapat disinkronkan. Coba muat ulang halaman.",
        );
      }
    };
    void load();
    const timer = window.setInterval(() => void load(), 10000);
    return () => {
      disposed = true;
      controller.abort();
      window.clearInterval(timer);
    };
  }, [taskId, organizationId, router]);

  return (
    <section className="overflow-hidden rounded-lg border border-line bg-white">
      <header className="flex flex-wrap items-center justify-between gap-2 border-b border-line px-4 py-3">
        <h2 className="text-sm font-semibold text-ink">Aktivitas Ranger & progres tugas</h2>
        <span className="text-xs text-muted">Sinkronisasi sekitar 10 detik</span>
      </header>
      <div className="p-4">
        {snapshot && (
          <p className="mb-3 text-xs text-muted">
            Status saat ini: <strong className="text-ink">{snapshot.task.status}</strong>
          </p>
        )}
        {!snapshot && !error && (
          <p className="text-xs text-muted">Memuat aktivitas tugas…</p>
        )}
        {error && <p role="alert" className="mb-3 text-xs text-danger">{error}</p>}
        {snapshot && snapshot.items.length === 0 && (
          <p className="text-xs text-muted">Belum ada transisi atau temuan yang tersimpan.</p>
        )}
        {snapshot && snapshot.items.length > 0 && (
          <ol className="space-y-3">
            {snapshot.items.map((item) => (
              <li key={`${item.kind}:${item.id}`} className="rounded-md border border-line bg-bg p-3">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <p className="text-[13px] font-semibold text-ink">
                    {item.kind === "finding" ? "Laporan Ranger" : "Perubahan status tugas"}
                  </p>
                  <time className="text-xs text-muted" dateTime={item.occurredAt}>
                    {new Intl.DateTimeFormat("id-ID", {
                      timeZone: "Asia/Makassar",
                      day: "2-digit", month: "short", year: "numeric",
                      hour: "2-digit", minute: "2-digit", hourCycle: "h23",
                    }).format(new Date(item.occurredAt))} WITA
                  </time>
                </div>
                {item.kind === "transition" && (
                  <p className="mt-1 text-xs text-muted">
                    {item.transitionId} · {item.title}
                  </p>
                )}
                <p className="mt-2 whitespace-pre-wrap text-[13px] leading-6 text-ink">
                  {item.description}
                </p>
              </li>
            ))}
          </ol>
        )}
      </div>
    </section>
  );
}
